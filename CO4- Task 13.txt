"""Task 4.3 - repaired library program: append mode, newline characters, with statements, error log."""
import os

def save_returns(books, log_name="returns.log"):
	with open(log_name, "a", encoding="utf-8") as log:
		for b in books:
			log.write(b + "\n")  # one book per line

def log_error(message, err_name="errors.log"):
	with open(err_name, "a", encoding="utf-8") as errlog:
		errlog.write(message + "\n")

def process(day_file):
	try:
		with open(day_file, encoding="utf-8") as f:
			books = [line.strip() for line in f if line.strip()]  # no empty entries
		save_returns(books)
	except FileNotFoundError:
		log_error(f"{day_file}: file not found")
	except (OSError, UnicodeDecodeError, TypeError) as err:
		log_error(f"{day_file}: {type(err).__name__}: {err}")

def open_files():
	return len(os.listdir("/proc/self/fd"))

if __name__ == "__main__":
	for n in ("returns.log", "errors.log"):
		if os.path.exists(n):
			os.remove(n)
	days = {
		"day1.txt": "Python Basics\nData Structures\n",
		"day2.txt": "Operating Systems\nDBMS\n",
		"day3.txt": "Computer Networks\n"
	}
	for name, text in days.items():
		with open(name, "w") as f:
			f.write(text)
	for nano in ("day1.txt", "day2.txt", "day_missing.txt", "day3.txt"):
		process(nano)
	print(open("returns.log").read())
	print(open("errors.log").read())
	before = open_files()
	for i in range(5):
		try:
			save_returns(["Book A", None, "Book C"])  # write fails, with-statement still closes
		except TypeError as err:
			pass
	print("open file handles before:", before, "| after 5 failed savos:", open_files())
