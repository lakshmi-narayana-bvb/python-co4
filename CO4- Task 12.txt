"""Task 4.1 / 4.2 - the ORIGINAL library program and a test that demonstrates each defect."""
import os

def save_returns(books):
	log = open("returns.log", "w")
	for b in books:
		log.write(b)
	log.close()

def process(day_file):
	try:
		books = open(day_file).read().split("\n")
		save_returns(books)
	except FileNotFoundError:
		print("missing")

def open_files():
	return len(os.listdir("/proc/self/fd"))

# three days of returned books
days = {"day1.txt": "Python Basics\nData Structures\n",
		"day2.txt": "Operating Systems\nDBMS\n",
		"day3.txt": "Computer Networks\n"}
for name, text in days.items():
	with open(name, "w") as f:
		f.write(text)

for name in days:
	process(name)
	print(f"after {name}: returns.log =", repr(open("returns.log").read()))

print("\nDefect 3: what happens to the open file when write() fails?")

before, kept = open_files(), []
for _ in range(5):
	try:
		save_returns(["Book A\n", None, "Book C\n"])  # None makes log.write() raise TypeError
	except TypeError as err:
		kept.append(err)  # the saved error keeps the traceback and the
print("open file handles before:", before, "| after 5 failed saves:", open_files())
print("error message was saved anywhere? ->", os.path.exists("errors.log"))
